{{- /* Előző/következő bejegyzés a saját szekción és nyelven belül, tisztán dátum szerint. Bal oldalt a régebbi, jobb oldalt az újabb. A .PrevInSection/.NextInSection a weight mezőt is figyelembe veszi, ezért itt saját, dátum szerinti lista dönt. */ -}} {{- $pages := .CurrentSection.RegularPages.ByDate -}} {{- $older := false -}} {{- $newer := false -}} {{- range $i, $p := $pages -}} {{- if eq $p $ -}} {{- if gt $i 0 }}{{ $older = index $pages (sub $i 1) }}{{ end -}} {{- if lt (add $i 1) (len $pages) }}{{ $newer = index $pages (add $i 1) }}{{ end -}} {{- end -}} {{- end -}} {{- if or $older $newer }}
{{- with $older }} ← {{ i18n "post_older" }} {{ .LinkTitle }} {{- else }} {{- end }} {{- with $newer }} {{ i18n "post_newer" }} → {{ .LinkTitle }} {{- end }}
{{- end }}
